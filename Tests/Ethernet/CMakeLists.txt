cmake_minimum_required(VERSION 3.20)
project(BetterCastEthernetTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
set(SENDER_DIR ../../Sources/BetterCastReceiverDesktop/sender)
add_executable(transport-tests TransportTests.cpp ${SENDER_DIR}/NetworkSender.cpp ${SENDER_DIR}/NetworkSender.h)
target_compile_definitions(transport-tests PRIVATE BETTERCAST_ETHERNET_TEST=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(transport-tests PRIVATE ${SENDER_DIR})
target_link_libraries(transport-tests PRIVATE Qt6::Core Qt6::Network)
if(WIN32)
    target_compile_definitions(transport-tests PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000003)
    target_link_libraries(transport-tests PRIVATE ws2_32)
endif()
find_path(AVCODEC_INCLUDE libavcodec/avcodec.h REQUIRED)
find_library(AVCODEC_LIB avcodec REQUIRED)
find_library(AVUTIL_LIB avutil REQUIRED)
add_executable(encoder-tests EncoderTests.cpp ${SENDER_DIR}/VideoEncoderFF.cpp ${SENDER_DIR}/VideoEncoderFF.h)
target_compile_definitions(encoder-tests PRIVATE BETTERCAST_ETHERNET_TEST=1)
target_include_directories(encoder-tests PRIVATE ${SENDER_DIR} ${AVCODEC_INCLUDE})
target_link_libraries(encoder-tests PRIVATE Qt6::Core ${AVCODEC_LIB} ${AVUTIL_LIB})
